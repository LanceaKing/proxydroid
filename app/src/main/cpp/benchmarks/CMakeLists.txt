cmake_minimum_required(VERSION 3.22.1)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(proxydroid-benchmark-server C)
else()
    add_executable(tls-benchmark client.c)
    target_compile_definitions(tls-benchmark PRIVATE _GNU_SOURCE)
    target_link_libraries(tls-benchmark PRIVATE redsocks-core ProxyDroid::TLS Threads::Threads m)
    set_target_properties(tls-benchmark PROPERTIES C_STANDARD 11 BUILD_RPATH "$ORIGIN/../redsocks;$ORIGIN")
endif()
find_package(Threads REQUIRED)
if(ANDROID AND TLS_BACKEND STREQUAL "openssl")
    add_executable(tls-bench-server server.c)
    target_compile_definitions(tls-bench-server PRIVATE _GNU_SOURCE)
    target_link_libraries(tls-bench-server PRIVATE ProxyDroid::TLS Threads::Threads)
elseif(NOT ANDROID)
    if(NOT TARGET OpenSSL::SSL)
        find_package(OpenSSL 1.1.1 REQUIRED)
    endif()
    add_executable(tls-bench-server server.c)
    target_compile_definitions(tls-bench-server PRIVATE _GNU_SOURCE)
    target_link_libraries(tls-bench-server PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
    if(BUILD_TESTING AND NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_test(NAME tls-benchmark-loopback COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/smoke.py" --server "$<TARGET_FILE:tls-bench-server>"
            --client "$<TARGET_FILE:tls-benchmark>")
        set_tests_properties(tls-benchmark-loopback PROPERTIES TIMEOUT 120)
    endif()
endif()
