cmake_minimum_required(VERSION 3.22.1)
project(proxydroid C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
foreach(dependency cntlm redsocks libevent)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dependency}/CMakeLists.txt")
        message(FATAL_ERROR "Missing ${dependency} submodule; run git submodule update --init --recursive")
    endif()
endforeach()
include(cmake/tls.cmake)
# Use upstream's build and generated configuration without modifying its source.
if(DEFINED EVENT__LIBRARY_TYPE AND NOT EVENT__LIBRARY_TYPE STREQUAL "STATIC")
    message(FATAL_ERROR "ProxyDroid requires EVENT__LIBRARY_TYPE=STATIC")
endif()
if(EVENT__DISABLE_THREAD_SUPPORT)
    message(FATAL_ERROR "ProxyDroid requires libevent thread support")
endif()
set(EVENT__LIBRARY_TYPE STATIC CACHE STRING "libevent library type")
set(EVENT__DISABLE_OPENSSL OFF CACHE BOOL "Build libevent's selected TLS transport")
set(EVENT__DISABLE_TESTS ON CACHE BOOL "Disable upstream libevent tests")
set(EVENT__DISABLE_REGRESS ON CACHE BOOL "Disable upstream libevent regression suite")
set(EVENT__DISABLE_SAMPLES ON CACHE BOOL "Disable upstream libevent samples")
set(EVENT__DISABLE_BENCHMARK ON CACHE BOOL "Disable upstream libevent benchmarks")
function(proxydroid_add_libevent)
    # libevent 2.1.13's minimum predates CMake 4; keep this caller-side policy
    # adjustment and OpenSSL discovery bridge out of the dependency's source.
    set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
    list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
    add_subdirectory(libevent EXCLUDE_FROM_ALL)
endfunction()
proxydroid_add_libevent()
add_subdirectory(cntlm)
set(REDSOCKS_SOURCE_INFO_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../scripts/native_sources.py")
add_subdirectory(redsocks)
if(BUILD_TESTING AND NOT ANDROID)
    add_executable(tls-contract-test cmake/tls-contract-test.c)
    target_link_libraries(tls-contract-test PRIVATE event_openssl ProxyDroid::TLS)
    add_test(NAME tls-contract COMMAND tls-contract-test)
endif()
option(PROXYDROID_BUILD_BENCHMARKS "Build opt-in native TLS/engine benchmark drivers" OFF)
if(PROXYDROID_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()
